# Data Model (replaces "database")

All content is static, typed TypeScript in `src/data/`. Types live in `src/types/index.ts` — **that file is the source of truth**; the block below is the original design and may lag behind it.
The actual values are in [05-content](../05-content/).

## As implemented (Phase 2) — differences from the design below
| Area | Implementation |
|---|---|
| Images | `ImageRef = { src, srcSet?, width, height, alt }`. Generated by `npm run images` into `src/data/images.generated.ts`; data files call `image(key, alt)` / `captioned(key, alt, caption)` from `src/lib/images.ts`. A wrong key is a compile error. |
| Icons | Data stores lucide icon **names** typed as `IconName` (`src/lib/icons.ts` registry) — typos fail type-check. |
| `ISODate` | `YYYY`, `YYYY-MM` or `YYYY-MM-DD`. |
| Project | `techStack` is grouped: `{ label, items[] }[]`. Added `appName`, `dataset`, `metrics: { value, label }[]`, `relatedExperience`, `confidentialNote`. `screenshots` carry a `caption`. |
| Achievement | `related?: { label, to }` (internal link, e.g. to a certificate), `cover?`, `photos` with captions. |
| Experience | `orgFacts` → `facts` (org certifications *and* program facts like "332 hours"); added `testimonial`, `related`; `proof.credentialId`. `start` optional. |
| Stats | `Stat.prefix` added. |

```ts
// src/types/index.ts

export type ISODate = string; // "2026-07-29" or "2026-07" when day unknown

export interface LinkRef {
  label: string;
  href: string;
  kind: 'live' | 'github' | 'verify' | 'pdf' | 'linkedin' | 'email' | 'other';
}

export interface ImageRef {
  src: string;          // imported asset URL
  alt: string;
  width: number;
  height: number;
}

/* ---------- Profile ---------- */
export interface Profile {
  name: string;
  headline: string;               // static line under name
  roles: string[];                // word rotator
  objectiveShort: string;         // Home hero
  objectiveLong: string;          // About intro
  location: string;
  email: string;
  showPhone: boolean;             // false by default
  phoneMasked?: string;           // "+91 738 •••• 858" — the full number is never stored
  photo: ImageRef;                // portrait 4:5
  avatar: ImageRef;               // 1:1
  availability: string;           // "Open to Assistant Professor & AI/ML roles"
  openTo: string[];
  teachingPhilosophy?: string;
  socials: LinkRef[];
  resumeUrl: string;              // "/resume/Aditi-Singh-Resume.pdf"
  motto?: string;
  languages: { name: string; level: string }[];
}

/* ---------- About ---------- */
export interface Education {
  id: string;
  degree: string;                 // "MCA"
  degreeFull: string;             // "Master of Computer Applications"
  institution: string;
  location: string;
  start: number;                  // 2024
  end: number;                    // 2026
  score: { value: number; unit: '%' | 'CGPA' };
  status: 'Completed' | 'Pursuing';
  highlights?: string[];
}

export type SkillCategory =
  | 'AI / ML' | 'Programming' | 'Web' | 'Database' | 'BI & Tools' | 'Cloud';

export interface SkillGroup {
  category: SkillCategory;
  icon: string;                   // lucide icon name
  skills: string[];
}

export interface LearningItem {
  topic: string;
  note: string;
  evidence?: string;              // e.g. "Kaggle: Intro to Deep Learning (Jul 2026)"
}

export interface CompetencyGroup { title: string; items: string[]; }

export interface JourneyItem {
  when: string;                   // "Mar 2025", "2024", "Next →"
  title: string;
  detail: string;
  href?: string;
  upcoming?: boolean;             // the "Next →" node
}

/* ---------- Projects ---------- */
export interface Project {
  slug: string;
  title: string;
  tagline: string;
  category: 'AI / ML' | 'Data Analytics' | 'Web';
  context: string;                // "MCA Final Year Project"
  period?: string;
  cover?: ImageRef;
  summary: string;                // card text
  problem: string;
  solution: string;
  features: string[];
  pipeline?: string[];            // steps for the DrawPath diagram
  techStack: string[];
  screenshots: ImageRef[];
  cursorTour?: { x: number; y: number; label: string }[]; // % positions on cover
  beforeAfter?: { before: ImageRef; after: ImageRef; beforeLabel: string; afterLabel: string };
  results?: string[];             // only real, verified results
  futureScope?: string[];
  links: LinkRef[];
  relatedExperience?: string;     // experience slug (internship project)
  confidentialNote?: string;      // "Details of client work are confidential."
  featured: boolean;
}

/* ---------- Certificates ---------- */
export type CertCategory =
  | 'AI & ML' | 'Data Science' | 'Cloud' | 'Programming & Web' | 'Workshops' | 'Other';

export interface Certificate {
  slug: string;
  title: string;
  issuer: string;                 // "Kaggle"
  platform?: string;              // "Udemy" when issuer is the instructor
  instructors?: string[];
  category: CertCategory;
  issueDate?: ISODate;
  eventDate?: ISODate;            // workshops
  duration?: string;              // "19 hours", "332 hours"
  grade?: string;                 // "96.00%", "A1"
  credentialId?: string;
  verifyUrl?: string;
  image: ImageRef;                // generated webp
  fileUrl: string;                // "/certificates/<slug>.pdf|.jpeg"
  skills: string[];
  description: string;
  featured: boolean;              // goes into ShuffleDeck
}

/* ---------- Achievements ---------- */
export interface Achievement {
  slug: string;
  title: string;
  organisation: string;
  date?: ISODate;
  role?: string;                  // "Team Leader"
  category: 'Competition' | 'Academic' | 'Leadership' | 'Speaking';
  icon: string;
  summary: string;
  story: string[];                // paragraphs for detail
  metric?: { value: string; label: string }; // for NumberRoll
  skills: string[];
  proof?: { image: ImageRef; fileUrl: string };
  cover?: ImageRef;               // GOER stage photo
  photos?: (ImageRef & { caption: string })[];
}

/* ---------- Experience ---------- */
export interface Experience {
  slug: string;
  role: string;
  organisation: string;
  orgFacts?: string[];            // "CMMI Level 5", "ISO 9001:2015"
  type: 'Internship' | 'Training' | 'Academic';
  location: string;
  start?: ISODate;                // may be unknown (e.g. Anudip AWS)
  end?: ISODate;                  // undefined + start set = present
  summary: string;
  responsibilities: string[];
  skills: string[];
  learnings?: string[];
  compare?: { left: string; right: string; rows: [string, string][] }; // CompareColumns
  proof?: { image: ImageRef; fileUrl: string; credentialId?: string };
}

/* ---------- Home ---------- */
export interface Stat { value: number; suffix?: string; decimals?: number; label: string; }
export interface Ping { icon: string; text: string; href: string; }
```

## Data rules (enforced by tests — see [test plan](../07-testing/test-plan.md))
1. `slug` unique within each collection, kebab-case.
2. Every `image.src` and `fileUrl` resolves to an existing file.
3. Arrays sorted newest first (by `issueDate` / `start`) unless `featured` ordering is used.
4. Optional fields that are unknown are **omitted**, never set to `""` or `"TBD"`. The UI hides the row/label for an omitted field (owner fills it in later).
